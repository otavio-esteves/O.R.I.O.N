cmake_minimum_required(VERSION 3.22)
project(orion_llama LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set_property(GLOBAL PROPERTY JOB_POOLS orion_compile=2 orion_link=1)
set(CMAKE_JOB_POOL_COMPILE orion_compile)
set(CMAKE_JOB_POOL_LINK orion_link)
# A gitlink pins source and licenses; no source download or subprocess is used by the APK.
set(ORION_LLAMA_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../../../../third_party/llama.cpp")
if(NOT EXISTS "${ORION_LLAMA_SOURCE}/include/llama.h")
    message(FATAL_ERROR "Initialize the pinned source: git submodule update --init --recursive")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_AVX OFF CACHE BOOL "" FORCE)
set(GGML_AVX2 OFF CACHE BOOL "" FORCE)
set(GGML_FMA OFF CACHE BOOL "" FORCE)
set(GGML_F16C OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
add_subdirectory("${ORION_LLAMA_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/llama")
add_library(orion_llama SHARED orion_llama.cpp)
target_compile_options(orion_llama PRIVATE -Wall -Wextra -Werror)
target_link_options(orion_llama PRIVATE "-Wl,-z,max-page-size=16384")
target_link_libraries(orion_llama PRIVATE llama)
